(defun my/treemacs-ipynb-auto-init ()
  "Automatically initialize and convert empty .ipynb files when opened."
  (when (and (string-suffix-p ".ipynb" (buffer-file-name))
             (= (buffer-size) 0))
    (message "Initializing new Jupyter Notebook...")
    
    ;; 1. Insert raw JSON
    (insert spacemacs//ipynb-template-string)
    
    ;; 2. Kill the 'modified' state and save as RAW JSON first
    ;; We use write-region to bypass the code-cells save-hooks
    (write-region (point-min) (point-max) (buffer-file-name))
    (set-buffer-modified-p nil)

    ;; 3. Now convert the display to percent-format
    (goto-char (point-min))
    ))

(add-hook 'find-file-hook #'my/treemacs-ipynb-auto-init)

* Terminal IPython REPL

#+begin_src emacs-lisp :tangle yes
(require 'vterm)

(defvar jupyter-ipython-vterm-registry (make-hash-table :test #'equal)
  "Map Jupyter kernelspec names to their live IPython vterm buffers.")

(defvar-local jupyter-ipython-vterm-buffer nil
  "The IPython vterm associated with the current source buffer.")

(defvar-local jupyter-ipython-code-cells-routing nil
  "Whether code-cells evaluation is routed through terminal IPython.")

(defun jupyter-ipython-vterm-live-p (buffer)
  "Return non-nil when BUFFER contains a live IPython vterm."
  (and (buffer-live-p buffer)
       (with-current-buffer buffer
         (and (derived-mode-p 'vterm-mode)
              (process-live-p (get-buffer-process buffer))))))

(defun jupyter-ipython-vterm-command (kernelspec)
  "Return the terminal IPython command derived from KERNELSPEC."
  (let ((interpreter (car (append
                           (plist-get (jupyter-kernelspec-plist kernelspec)
                                      :argv)
                           nil))))
    (unless (and (stringp interpreter)
                 (not (string-empty-p interpreter)))
      (user-error "Kernelspec %s has no interpreter"
                  (jupyter-kernelspec-name kernelspec)))
    (list interpreter "-m" "IPython")))

(defun jupyter-ipython-vterm-environment (kernelspec)
  "Return a process environment that applies KERNELSPEC settings."
  (append (jupyter-process-environment kernelspec) process-environment))

(defun jupyter-ipython-vterm-buffer-name (kernelspec)
  "Return the vterm buffer name for KERNELSPEC."
  (format "*ipython[%s]*"
          (or (plist-get (jupyter-kernelspec-plist kernelspec) :display_name)
              (jupyter-kernelspec-name kernelspec))))

(defun jupyter-ipython-clear-associations (vterm-buffer)
  "Clear registry and source-buffer associations for VTERM-BUFFER."
  (maphash (lambda (name buffer)
             (when (eq buffer vterm-buffer)
               (remhash name jupyter-ipython-vterm-registry)))
           jupyter-ipython-vterm-registry)
  (dolist (buffer (buffer-list))
    (when (eq (buffer-local-value 'jupyter-ipython-vterm-buffer buffer)
              vterm-buffer)
      (with-current-buffer buffer
        (setq jupyter-ipython-vterm-buffer nil
              jupyter-ipython-code-cells-routing nil)))))

(defun jupyter-ipython-vterm-exited (buffer _event)
  "Clear associations when an IPython vterm BUFFER exits."
  (jupyter-ipython-clear-associations buffer))

(add-hook 'vterm-exit-functions #'jupyter-ipython-vterm-exited)

(defun jupyter-ipython-associated-vterm ()
  "Return the live IPython vterm associated with the current buffer."
  (if (jupyter-ipython-vterm-live-p jupyter-ipython-vterm-buffer)
      jupyter-ipython-vterm-buffer
    (when jupyter-ipython-vterm-buffer
      (jupyter-ipython-clear-associations jupyter-ipython-vterm-buffer))
    (user-error
     "No live IPython vterm is associated; run jupyter-repl-ipython or jupyter-ipython-associate-buffer")))

(defun jupyter-ipython-associate-buffer (vterm-buffer)
  "Associate the current source buffer with live VTERM-BUFFER."
  (interactive
   (let (candidates)
     (maphash (lambda (name buffer)
                (when (jupyter-ipython-vterm-live-p buffer)
                  (push name candidates)))
              jupyter-ipython-vterm-registry)
     (unless candidates
       (user-error "No live IPython vterm; run jupyter-repl-ipython first"))
     (list (gethash (completing-read "IPython kernel: " candidates nil t)
                    jupyter-ipython-vterm-registry))))
  (unless (jupyter-ipython-vterm-live-p vterm-buffer)
    (user-error "The selected IPython vterm is no longer live"))
  (setq-local jupyter-ipython-vterm-buffer vterm-buffer
              jupyter-ipython-code-cells-routing t)
  (message "Associated %s with %s" (buffer-name)
           (buffer-name vterm-buffer)))

(defun jupyter-ipython-source-string (beg end)
  "Return source from BEG to END without code-cell delimiters."
  (replace-regexp-in-string
   "^[[:space:]]*#[[:space:]]*%%.*\\(?:\n\\|\\'\\)"
   ""
   (buffer-substring-no-properties beg end)))

(defun jupyter-ipython-send-region (beg end)
  "Send the source region from BEG to END to its associated IPython vterm."
  (interactive "r")
  (let ((vterm-buffer (jupyter-ipython-associated-vterm))
        (source (jupyter-ipython-source-string beg end)))
    (with-current-buffer vterm-buffer
      (vterm-send-string source t)
      (vterm-send-return))))

(defun jupyter-ipython-send-cell ()
  "Send the current code cell to its associated IPython vterm."
  (interactive)
  (require 'code-cells)
  (pcase-let ((`(,beg ,end) (code-cells--bounds 1 nil t)))
    (jupyter-ipython-send-region beg end)))

(defun jupyter-ipython-send-whole-buffer ()
  "Send the current buffer to its associated IPython vterm."
  (interactive)
  (jupyter-ipython-send-region (point-min) (point-max)))

(defun jupyter-ipython-pop-to-vterm ()
  "Display the current buffer's associated IPython vterm."
  (interactive)
  (pop-to-buffer (jupyter-ipython-associated-vterm)))

(defun jupyter-ipython-interrupt ()
  "Send an interrupt to the current buffer's associated IPython vterm."
  (interactive)
  (with-current-buffer (jupyter-ipython-associated-vterm)
    (vterm-send-key "c" nil nil t)))

(defun jupyter-ipython-enable-code-cells-routing ()
  "Install terminal-IPython routing for the current Python buffer.
The route becomes active only after explicit vterm association."
  (setq-local code-cells-eval-region-commands
              (cons '(jupyter-ipython-code-cells-routing
                      . jupyter-ipython-send-region)
                    (assq-delete-all
                     'jupyter-ipython-code-cells-routing
                     (copy-tree code-cells-eval-region-commands)))))

(require 'code-cells)
(add-hook 'python-mode-hook #'jupyter-ipython-enable-code-cells-routing)
(add-hook 'python-ts-mode-hook #'jupyter-ipython-enable-code-cells-routing)

(defun jupyter-repl-ipython (kernelspec)
  "Start or visit terminal IPython for a selected Jupyter KERNELSPEC."
  (interactive (list (jupyter-completing-read-kernelspec
                      nil current-prefix-arg)))
  (let* ((source-buffer (current-buffer))
         (name (jupyter-kernelspec-name kernelspec))
         (existing (gethash name jupyter-ipython-vterm-registry)))
    (let ((vterm-buffer
           (if (jupyter-ipython-vterm-live-p existing)
               (progn
                 (pop-to-buffer existing)
                 existing)
             (when existing
               (jupyter-ipython-clear-associations existing))
             (let* ((command (jupyter-ipython-vterm-command kernelspec))
                    (process-environment
                     (jupyter-ipython-vterm-environment kernelspec))
                    (vterm-shell (mapconcat #'shell-quote-argument command " "))
                    (buffer (vterm-other-window
                             (jupyter-ipython-vterm-buffer-name kernelspec))))
               (puthash name buffer jupyter-ipython-vterm-registry)
               buffer))))
      (when (called-interactively-p 'interactive)
        (with-current-buffer source-buffer
          (jupyter-ipython-associate-buffer vterm-buffer)))
      vterm-buffer)))
